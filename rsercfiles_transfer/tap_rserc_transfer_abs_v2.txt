#!/bin/bash
###############################################################################
# Script Name   : tap_rserc_transfer_abs_v2.sh
# Converted From: TAP_RSERCFILES_TRANSFER.COM (VMS DCL) — NEW VERSION
# Description   : Transfers generated RSERC and MRLOG files from the TAP
#                 system to the ABS system via SFTP. Runs continuously,
#                 batching up to 50 files of each type per cycle. Uses a
#                 temporary file extension (.sftp_tmp_rs) during transfer
#                 to prevent partial pickup on the remote side. Includes
#                 self-recovery from previous failed transfers.
#
# Environment   : Linux / Bash 4+ / RHEL 9
# Dependencies  : sftp (OpenSSH), SSH key-based auth to ABS server
#
# Original Author : Mahalakshmi / Arul Mani (VMS legacy)
# Converted By    : VMS-to-Linux conversion
#
# Version History:
#   v1  — Initial conversion from OLD COM script
#   v2  — Updated to match NEW COM script changes:
#         - Added OLD_SER pre-check in self_recovery
#         - Added MAIN_RECOVERY section with external recovery job call
#         - Changed remote directory to /DISK$BME_APPS/BME/TAP/INPUT/
#         - RSERC files renamed to .BCKUP instead of deleted
#         - RSERC remote rename now inline in SFTP batch (no separate step)
#         - MRLOG files use .DAT extension directly (no .sftp_tmp_rs)
#         - Added backup archival step in main_loop
#         - Added pre-rename of RSERC to .OLD_SER before main loop
#         - Added hour=23 auto-shutdown check
#         - External recovery script (tranfer_rserc_recovery.sh) ACTIVE
#
# VMS Sections Mapped:
#   SAVE_ENVIRONMENT       -> Configuration section + lock file
#   CHECK_INSTANCES        -> acquire_lock()  (flock-based)
#   Logical checks         -> validate_environment()
#   HOUSE_KEEP             -> housekeeping()
#   EXTRACT_SFTP           -> read_sftp_config()
#   SELF_RECOVERY          -> self_recovery()
#   MAIN_RECOVERY          -> main_recovery()           [NEW v2]
#   MAIN_LOOP              -> main_loop()
#   RSERC_FILE_COUNT       -> collect_rserc_files()
#   MRLOG_FILE_COUNT       -> collect_mrlog_files()
#   SFTP_PROCESS           -> sftp_transfer()
#   FINAL_CHECK            -> should_shutdown()
#   ERROR                  -> error_exit()
#   CLEAN_FINISH / EXIT    -> cleanup_and_exit()
#
# Prerequisites:
#   1. SSH key-based authentication configured for SFTP to ABS server
#   2. Config file RSERC_SFTP.CFG in TAP_CFG_DIR (line1=user, line2=host)
#   3. All directory environment variables set (see CONFIGURATION below)
#   4. RSERC_TRANS_SHUTDOWN env var or flag file for graceful shutdown
#   5. Write access to SFTP_TMP_DIR, TAP_DAT_DIR, TAP_LOG_DIR
#   6. [NEW v2] tranfer_rserc_recovery.sh in TAP_COM_DIR
#   7. [NEW v2] FCS_ARCHIVE_DIR for backup archival
###############################################################################
 
set -o pipefail
 
###############################################################################
#                       CONFIGURATION
###############################################################################
 
# --- Directory mappings (VMS logical -> Linux path) ---
# Defaults can be overridden via environment variables (for testing/deployment)
TAP_CFG_DIR="${TAP_CFG_DIR:-/app/tap/R53_TAPLIVE/TAP/CFG}"
TAP_DAT_DIR="${TAP_DAT_DIR:-/data/tap/R53_TAPLIVE/TAP/DAT}"
TAP_LOG_DIR="${TAP_LOG_DIR:-/data/tap/R53_TAPLIVE/TAP/LOG}"
TAP_SH_DIR="${TAP_SH_DIR:-/app/tap/R53_TAPLIVE/TAP/SH}"
FCS_RSERC_DIR="${FCS_RSERC_DIR:-/data/tap/R53_TAPLIVE/TAP/FCS_RSERC}"
TAP_RSERC_DIR="${TAP_RSERC_DIR:-/data/tap/R53_TAPLIVE/TAP/RSERC}"
SFTP_TMP_DIR="${SFTP_TMP_DIR:-/data/tap/R53_TAPLIVE/TAP/DAT/SFTP_TMP}"
 
## [NEW v2] Archive directory for RSERC backups (VMS: DISK$CALL_DATA2:[TAP.OB.FCS_ARCHIVE])
TAP_FCS_ARCHIVE="${TAP_FCS_ARCHIVE:-/data/tap/R53_TAPLIVE/TAP/FCS_ARCHIVE}"
 
# --- Shutdown controls (VMS: RSERC_TRANS_SHUTDOWN logical, TAP_CLOSEDOWN_ALL) ---
RSERC_TRANS_SHUTDOWN="${RSERC_TRANS_SHUTDOWN:-N}"
TAP_CLOSEDOWN_ALL="${TAP_CLOSEDOWN_ALL:-N}"
 
# --- SFTP config file (VMS: TAP_CFG_DIR:RSERC_SFTP.CFG) ---
# SFTP credentials (DEST_USERNAME, DEST_HOSTNAME) are ALWAYS read from this
# config file via read_sftp_config(). They are NEVER hardcoded in the script.
SFTP_CFG_FILE="${TAP_CFG_DIR}/RSERC_SFTP.CFG"
 
# --- Batch size limit (VMS: .GT. 50) ---
MAX_BATCH_SIZE=50
 
## [NEW v2] Remote directory on ABS server
## OLD: REMOTE_DIR="xi_dat"
## NEW: Changed to absolute path per new COM script
REMOTE_DIR="/home/brabhav/BME/TAP/INPUT"
 
# --- Log file ---
SCRIPT_NAME="$(basename "$0" .sh)"
LOG_FILE="${TAP_LOG_DIR}/${SCRIPT_NAME}_$(date '+%Y%m%d').log"
 
# --- Lock file for single-instance enforcement ---
LOCK_FILE="/tmp/${SCRIPT_NAME}.lock"
 
# --- Flag file for in-progress tracking (VMS: SFTP_ABS_IN_PROGRESS.FLAG) ---
FLAG_FILE="${TAP_DAT_DIR}/SFTP_ABS_IN_PROGRESS.FLAG"
 
# --- Guard against double cleanup ---
_CLEANUP_DONE=0
 
# --- File descriptor for lock ---
LOCK_FD=9
 
###############################################################################
#                       FUNCTIONS
###############################################################################
 
# ---- Logging helper (VMS: @TAP_COM_DIR:TAPLOG_MESS + WSO) ----
log_msg() {
    local msg="$1"
    local dttm
    dttm=$(date '+%d-%b-%Y %H:%M:%S')
    echo "${dttm} - ${msg}" | tee -a "${LOG_FILE}"
}
 
# ---- Error exit (VMS: ERROR section) ----
error_exit() {
    local error_text="$1"
    local phase="${2:-UNKNOWN}"
 
    log_msg "ERROR: *** ${SCRIPT_NAME} - ${phase}, ${error_text}"
 
    # VMS: REQUEST/REPLY/TO=OPER — log to syslog as substitute
    logger -t "${SCRIPT_NAME}" "ERROR - ${phase}: ${error_text}"
 
    cleanup_and_exit 4
}
 
# ---- Cleanup (VMS: EXIT section + temp file removal) ----
cleanup_and_exit() {
    local exit_code="${1:-0}"
 
    if [ "${_CLEANUP_DONE}" -eq 1 ]; then
        exit "${exit_code}"
    fi
    _CLEANUP_DONE=1
 
    log_msg "${SCRIPT_NAME} exiting with code ${exit_code} at $(date '+%d-%b-%Y %H:%M:%S')"
 
    # Release lock
    if [ -n "${LOCK_FD}" ]; then
        eval "exec ${LOCK_FD}>&-" 2>/dev/null
    fi
 
    exit "${exit_code}"
}
 
# ---- Acquire exclusive lock (VMS: SET PROCESS /NAME= for instance check) ----
acquire_lock() {
    eval "exec ${LOCK_FD}>${LOCK_FILE}"
    if ! flock -n "${LOCK_FD}"; then
        echo "$(date '+%d-%b-%Y %H:%M:%S') - ${SCRIPT_NAME} exiting - process already running." | tee -a "${LOG_FILE}"
        exit 0
    fi
}
 
# ---- Validate environment (VMS: logical name checks) ----
validate_environment() {
    local phase="VALIDATE_ENVIRONMENT"
 
    ## [NEW v2] Added FCS_ARCHIVE_DIR to validation list
    for dir_var in TAP_CFG_DIR TAP_DAT_DIR TAP_LOG_DIR FCS_RSERC_DIR TAP_RSERC_DIR SFTP_TMP_DIR TAP_FCS_ARCHIVE; do
        local dir_val="${!dir_var}"
        if [ -z "${dir_val}" ]; then
            error_exit "*** Fatal: Environment variable ${dir_var} is not set." "${phase}"
        fi
        if [ ! -d "${dir_val}" ]; then
            error_exit "*** Fatal: Directory ${dir_val} (${dir_var}) does not exist." "${phase}"
        fi
    done
 
    if [ ! -f "${SFTP_CFG_FILE}" ]; then
        error_exit "ERROR WHILE SEARCHING FOR ${SFTP_CFG_FILE} FILE" "${phase}"
    fi
}
 
# ---- Housekeeping (VMS: HOUSE_KEEP — delete logs older than 7 days) ----
housekeeping() {
    log_msg "Housekeeping: removing log files older than 7 days"
    #find "${TAP_LOG_DIR}" -maxdepth 1 -name "${SCRIPT_NAME}.log.*" -mtime +7 -delete 2>/dev/null || true
    find "${TAP_LOG_DIR}" -maxdepth 1 -name "${SCRIPT_NAME}_*.log" -mtime +7 -delete 2>/dev/null || true
 
    ## [NEW v2] Placeholder for .BCKUP cleanup (VMS: DELETE/BEFORE="TODAY-3-00" FCS_RSERC_DIR:RSERC%%%%%%.BCKUP;*)
    ## Uncomment when backup retention policy is confirmed:
    # log_msg "Housekeeping: removing BCKUP files older than 3 days"
    # find "${FCS_RSERC_DIR}" -maxdepth 1 -name 'RSERC??????_*.BCKUP' -mtime +3 -delete 2>/dev/null || true
}
 
# ---- Read SFTP config (VMS: EXTRACT_SFTP — read username and hostname) ----
# Reads DEST_USERNAME and DEST_HOSTNAME from RSERC_SFTP.CFG
# Credentials are NEVER hardcoded — always read from this config file
read_sftp_config() {
    local phase="EXTRACT_SFTP"
 
    if [ ! -f "${SFTP_CFG_FILE}" ]; then
        error_exit "SFTP config file not found: ${SFTP_CFG_FILE}" "${phase}"
    fi
 
    # VMS: line 1 = username, line 2 = hostname
    {
        read -r DEST_USERNAME
        read -r DEST_HOSTNAME
    } < "${SFTP_CFG_FILE}"
 
    if [ -z "${DEST_USERNAME}" ] || [ -z "${DEST_HOSTNAME}" ]; then
        error_exit "SFTP config file is incomplete (need username and hostname)" "${phase}"
    fi
 
    log_msg "SFTP target: ${DEST_USERNAME}@${DEST_HOSTNAME}"
}
 
# ---- [NEW v2] Main Recovery (VMS: MAIN_RECOVERY section) ----
# Called when SFTP_ABS_IN_PROGRESS.FLAG is found during self_recovery
# 1. Renames current RSERC .DAT files to .OLD_SER (safety backup)
# 2. Calls external recovery script (tranfer_rserc_recovery.sh)
# 3. Loops back to self_recovery for re-verification
main_recovery() {
    local phase="MAIN_RECOVERY"
    log_msg "MAIN_RECOVERY: Unrecovered previous failure found"
 
    # Rename current RSERC *.DAT to *.OLD_SER for safety
    # (VMS: RENAME/LOG FCS_RSERC_DIR:RSERC*.DAT;*/BEF *.OLD_SER;*)
    log_msg "Renaming RSERC .DAT files to .OLD_SER for recovery"
    for dat_file in "${FCS_RSERC_DIR}"/RSERC*.DAT; do
        if [ -f "${dat_file}" ]; then
            local base
            base=$(basename "${dat_file}" .DAT)
            mv -v "${dat_file}" "${FCS_RSERC_DIR}/${base}.OLD_SER" 2>&1 | tee -a "${LOG_FILE}"
        fi
    done
 
    ## =========================================================================
    ## [NEW v2] EXTERNAL SCRIPT CALL — TRANFER_RSERC_RECOVERY
    ## =========================================================================
    ## VMS ORIGINAL:
    ##   SUBMIT/LOG=DSA10:[TAP.LOG]TRANFER_RSERC_RECOVERY.LOG
    ##     DISK$CALL_DATA:[TAP.COM]TRANFER_RSERC_RECOVERY.COM;
    ##     /NOTIFY/USER=TAPLIVE/NAME=TRANFER_RSERC_RECOVERY
    ##   SYNCHRONIZE TRANFER_RSERC_RECOVERY
    ##
    ## Linux equivalent: Run the converted recovery script synchronously.
    ## SFTP credentials are read from RSERC_SFTP.CFG inside the recovery
    ## script — no hardcoded usernames or hostnames.
    ## =========================================================================
    local recovery_script="${TAP_SH_DIR}/tranfer_rserc_recovery.sh"
 
    if [ ! -f "${recovery_script}" ]; then
        error_exit "Recovery script not found: ${recovery_script}" "${phase}"
    fi
 
    log_msg "Submitting recovery job: ${recovery_script}"
    if ! bash "${recovery_script}" >> "${TAP_LOG_DIR}/tranfer_rserc_recovery.log" 2>&1; then
        error_exit "Recovery job TRANFER_RSERC_RECOVERY failed" "${phase}"
    fi
 
    log_msg "Recovery job completed — looping back to self_recovery for re-check"
    # VMS: GOTO SELF_RECOVERY — loop back to re-verify everything is clean
    self_recovery
    return $?
}
 
# ---- Self-recovery (VMS: SELF_RECOVERY section) ----
self_recovery() {
    local phase="SELF_RECOVERY"
    log_msg "Checking for recovery from previous failure"
 
    ## [NEW v2] Pre-check: Look for leftover .OLD_SER files from a previous failed recovery
    local old_ser_count
    old_ser_count=$(find "${FCS_RSERC_DIR}" -maxdepth 1 -name 'RSERC*.OLD_SER' 2>/dev/null | wc -l)
    if [ "${old_ser_count}" -gt 0 ]; then
        log_msg "RECOVERY FAILED AND OLD RSERC'S FOUND, DROPPING ALERT MAIL TO L2"
        error_exit "OLD RSERC'S FOUND IN TAP RECOVERY NOT SUCCESSFULL - CALLOUT FOR L2" "${phase}"
    fi
 
    ## [NEW v2] SFTP_TMP_DIR cleanup removed per new COM (commented out in VMS)
    # rm -f "${SFTP_TMP_DIR:?}"/* 2>/dev/null || true
 
    # Check if a previous transfer was in progress
    if [ ! -f "${FLAG_FILE}" ]; then
        log_msg "No recovery needed"
        return 0
    fi
 
    ## [NEW v2] When flag file is found, go to MAIN_RECOVERY
    log_msg "Found in-progress flag — performing recovery via MAIN_RECOVERY"
    main_recovery
    return $?
}
 
# ---- Collect RSERC files for transfer ----
# [NEW v2] Changes:
#   - Remote rename is now INLINE in SFTP batch
#   - Source files RENAMED to .BCKUP instead of DELETED
collect_rserc_files() {
    RSERC_COUNT=0
    local sftp_fd="$1"
    local delete_fd="$2"
 
    while IFS= read -r -d '' rserc_file; do
        if [ "${RSERC_COUNT}" -ge "${MAX_BATCH_SIZE}" ]; then
            break
        fi
 
        local filename
        filename=$(basename "${rserc_file}")
        local name="${filename%.*}"
        local tmp_name="${name}.sftp_tmp_rs"
 
        if cp "${rserc_file}" "${SFTP_TMP_DIR}/${tmp_name}"; then
            ## [NEW v2] Upload AND rename in the SAME SFTP batch (inline)
            echo "put ${tmp_name}" >> "${sftp_fd}"
            echo "rename ${tmp_name} ${filename}" >> "${sftp_fd}"
 
            ## [NEW v2] RENAME source to .BCKUP instead of DELETE
            echo "mv \"${rserc_file}\" \"${FCS_RSERC_DIR}/${name}.BCKUP\"" >> "${delete_fd}"
 
            RSERC_COUNT=$((RSERC_COUNT + 1))
            log_msg "Staged RSERC file: ${filename}"
        else
            log_msg "WARNING: Failed to copy ${rserc_file} to staging"
        fi
done < <(find "${FCS_RSERC_DIR}" -maxdepth 1 -name 'RSERC??????_*.DAT' -print0 2>/dev/null)
}
 
# ---- Collect MRLOG files for transfer ----
# [NEW v2] Changes:
#   - MRLOG files now use .DAT extension directly (no .sftp_tmp_rs)
#   - No remote rename needed
collect_mrlog_files() {
    MRLOG_COUNT=0
    local sftp_fd="$1"
    local delete_fd="$2"
 
    while IFS= read -r -d '' mrlog_file; do
        if [ "${MRLOG_COUNT}" -ge "${MAX_BATCH_SIZE}" ]; then
            break
        fi
 
        local filename
        filename=$(basename "${mrlog_file}")
        local name="${filename%.*}"
 
        ## [NEW v2] Use .DAT extension directly — no temporary extension
        local tmp_name="${name}.DAT"
 
        if cp "${mrlog_file}" "${SFTP_TMP_DIR}/${tmp_name}"; then
            ## [NEW v2] Upload only — no rename needed
            echo "put ${tmp_name}" >> "${sftp_fd}"
            echo "rm -f \"${mrlog_file}\"" >> "${delete_fd}"
            MRLOG_COUNT=$((MRLOG_COUNT + 1))
            log_msg "Staged MRLOG file: ${filename}"
        else
            log_msg "WARNING: Failed to copy ${mrlog_file} to staging"
        fi
done < <(find "${TAP_RSERC_DIR}" -maxdepth 1 -name 'MRLOG??????_*.DAT' -print0 2>/dev/null)
 
}
 
# ---- SFTP transfer process ----
# [NEW v2] Separate rename SFTP call REMOVED — rename is inline
sftp_transfer() {
    local phase="SFTP_PROCESS"
    local sftp_batch="$1"
    local delete_script="$2"
 
    log_msg "SFTP PROCESS FOR TRANSFER OF RSERC AND MRLOG FILES STARTED AT $(date '+%d-%b-%Y %H:%M:%S')"
 
    touch "${FLAG_FILE}"
 
    # Step 1: Upload files via SFTP (includes inline rename for RSERC)
    if ! sftp -b "${sftp_batch}" "${DEST_USERNAME}@${DEST_HOSTNAME}"; then
        error_exit "ERROR WHILE TRANSFERRING THE RSERCS AND MRLOG FILES FROM TAP TO ABS - SFTP PROCESS" "${phase}"
    fi
    rm -f "${sftp_batch}"
 
    ## [NEW v2] Separate rename SFTP call REMOVED — rename is inline in upload batch
 
    # Clean staging directory
    rm -f "${SFTP_TMP_DIR:?}"/* 2>/dev/null || true
 
    # Step 2: Run local cleanup script (rename RSERC to .BCKUP / delete MRLOG)
    if ! bash "${delete_script}"; then
        error_exit "ERROR WHILE DELETING SFTPD FILES FROM XI_DAT DIRECTORY" "${phase}"
    fi
    rm -f "${delete_script}"
 
    rm -f "${FLAG_FILE}"
 
    log_msg "SFTP PROCESS FOR TRANSFER OF RSERC AND MRLOG FILES COMPLETED AT $(date '+%d-%b-%Y %H:%M:%S')"
}
 
# ---- Check if shutdown is requested ----
should_shutdown() {
    if [ "${RSERC_TRANS_SHUTDOWN}" = "Y" ]; then
        log_msg "Shutdown requested via RSERC_TRANS_SHUTDOWN flag"
        return 0
    fi
 
    if [ -f "${TAP_DAT_DIR}/RSERC_TRANS_SHUTDOWN.FLAG" ]; then
        log_msg "Shutdown requested via flag file"
        return 0
    fi
 
    if [ "${TAP_CLOSEDOWN_ALL}" != "N" ]; then
        local current_time
        current_time=$(date '+%H:%M')
        if [[ "${current_time}" > "${TAP_CLOSEDOWN_ALL}" ]]; then
            log_msg "Past closedown time (${TAP_CLOSEDOWN_ALL}), shutting down"
            return 0
        fi
    fi
 
    ## [NEW v2] Auto-shutdown at hour 23 (11 PM)
    local current_hour
    current_hour=$(date '+%H')
    if [ "${current_hour}" = "23" ]; then
        log_msg "Hour is 23 — auto-shutting down"
        return 0
    fi
 
    return 1
}
 
# ---- [NEW v2] Pre-rename RSERC files to .OLD_SER ----
pre_rename_rserc() {
    local phase="PRE_RENAME_RSERC"
    log_msg "Pre-renaming existing RSERC .DAT files to .OLD_SER before main loop"
 
    for dat_file in "${FCS_RSERC_DIR}"/RSERC*.DAT; do
        if [ -f "${dat_file}" ]; then
            local base
            base=$(basename "${dat_file}" .DAT)
            mv -v "${dat_file}" "${FCS_RSERC_DIR}/${base}.OLD_SER" 2>&1 | tee -a "${LOG_FILE}"
        fi
    done
}
 
# ---- [NEW v2] Archive backup files ----
archive_backups() {
    local phase="ARCHIVE_BACKUPS"
 
    local bckup_count
    bckup_count=$(find "${FCS_RSERC_DIR}" -maxdepth 1 -name '*.BCKUP' 2>/dev/null | wc -l)
 
    if [ "${bckup_count}" -gt 0 ]; then
        log_msg "Archiving ${bckup_count} .BCKUP file(s) to ${TAP_FCS_ARCHIVE}"
 
        for bckup_file in "${FCS_RSERC_DIR}"/*.BCKUP; do
            if [ -f "${bckup_file}" ]; then
                local base
                base=$(basename "${bckup_file}" .BCKUP)
                if cp "${bckup_file}" "${TAP_FCS_ARCHIVE}/${base}.DAT"; then
                    rm -f "${bckup_file}"
                    log_msg "Archived and removed: ${bckup_file}"
                else
                    log_msg "WARNING: Failed to archive ${bckup_file}"
                fi
            fi
        done
    fi
}
 
# ---- Main transfer loop ----
main_loop() {
    local phase="MAIN_LOOP"
 
    ## [NEW v2] Pre-rename RSERC files to .OLD_SER before entering loop
    pre_rename_rserc
 
    while true; do
 
        ## [NEW v2] Archive previous cycle's .BCKUP files
        archive_backups
 
        local sftp_batch="${TAP_DAT_DIR}/sftp_ctrl_file.dat"
        local delete_script="${TAP_DAT_DIR}/delete_sftpd_files_tmp.sh"
 
        ## [NEW v2] Remote directory changed to absolute path
        cat > "${sftp_batch}" <<EOF
lcd ${SFTP_TMP_DIR}
cd ${REMOTE_DIR}
EOF
 
        cat > "${delete_script}" <<'EOF'
#!/bin/bash
# Auto-generated: rename/delete source files after successful SFTP transfer
EOF
 
        ## [NEW v2] No separate rename batch — rename is inline in sftp_batch
        collect_rserc_files "${sftp_batch}" "${delete_script}"
        collect_mrlog_files "${sftp_batch}" "${delete_script}"
 
        if [ "${RSERC_COUNT}" -gt 0 ] || [ "${MRLOG_COUNT}" -gt 0 ]; then
            log_msg "Batch: ${RSERC_COUNT} RSERC files, ${MRLOG_COUNT} MRLOG files"
 
            echo "exit" >> "${sftp_batch}"
 
            sftp_transfer "${sftp_batch}" "${delete_script}"
        else
            rm -f "${sftp_batch}" "${delete_script}"
 
            log_msg "No files to transfer, waiting 10 minutes"
            sleep 600
        fi
 
        if should_shutdown; then
            break
        fi
    done
}
 
###############################################################################
#                       MAIN
###############################################################################
 
main() {
    trap 'error_exit "Caught signal, aborting" "SIGNAL"' SIGTERM SIGINT
    trap cleanup_and_exit EXIT
 
    log_msg "${SCRIPT_NAME} has started at $(date '+%d-%b-%Y %H:%M:%S')"
 
    acquire_lock
    validate_environment
    housekeeping
 
    # Read SFTP credentials from config file (never hardcoded)
    read_sftp_config
 
    self_recovery
    main_loop
 
    log_msg "${SCRIPT_NAME} has completed successfully at $(date '+%d-%b-%Y %H:%M:%S')"
}
